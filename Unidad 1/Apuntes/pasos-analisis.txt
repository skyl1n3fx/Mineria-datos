// Método para crear automáticamente un gráfico
def getGraficoBarras(label_x, label_y, titulo, dataframe, caracteristica):
  x_values = dataframe[caracteristica].unique()
  y_values = dataframe[caracteristica].value_counts().to_list()
  plt.bar(x_values, y_values)
  plt.xlabel(label_x)
  plt.ylabel(label_y)
  plt.show()
